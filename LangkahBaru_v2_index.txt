    function getDayOfYear() {
      const now = new Date();
      const start = new Date(now.getFullYear(), 0, 0);
      return Math.floor((now - start) / 86400000);

    function loadState() {
      try {
        const raw = localStorage.getItem(APP_CONFIG.storageKey);
        if (!raw) return defaultState();
        const merged = { ...defaultState(), ...JSON.parse(raw) };
        const today = getTodayKey();

        if (merged.lastActiveDate !== today) { merged.daysActive = (merged.daysActive || 1) + 1; merged.lastActiveDate = today; }
        if (merged.todayMoodDate !== today) { merged.todayMoodDate = today; merged.todayMood = ""; }
        if (merged.solatDate !== today) { merged.solatDate = today; merged.solatDone = { subuh: false, zohor: false, asar: false, maghrib: false, isyak: false }; }
        if (merged.tasbihTodayDate !== today) { merged.tasbihTodayDate = today; merged.tasbihTodayCount = 0; }
        if (merged.todayTasksDate !== today) { merged.todayTasksDate = today; merged.todayTasks = []; }
        if (merged.prayerBlocksDate !== today) {
          merged.prayerBlocksDate = today;
          Object.keys(merged.prayerBlocks).forEach(k => { merged.prayerBlocks[k].done = false; });
        }
        return merged;
      } catch (e) { return defaultState(); }
    }

    let audioCtx = null;
    function playSoftClick(freq = 520, duration = 0.045) {
      if (!state.tasbihSound) return;
      try {
        if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
        if (audioCtx.state === "suspended") audioCtx.resume();
        const osc = audioCtx.createOscillator();
        const gain = audioCtx.createGain();
        osc.type = "sine";
        osc.frequency.setValueAtTime(freq, audioCtx.currentTime);
        gain.gain.setValueAtTime(0.06, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + duration);
        osc.connect(gain);
        gain.connect(audioCtx.destination);
        osc.start();
        osc.stop(audioCtx.currentTime + duration);
      } catch (e) {}
    }

    function toggleTimer() {

          </div>
          <div class="add-row">
            <input type="text" id="roadmapItemInput" placeholder="Tambah langkah kecil..." maxlength="80" />
            <button class="btn btn-primary btn-sm" onclick="addRoadmapItem()">Tambah</button>
          </div>
        </div>
      `;
    }

    function saveRoadmapField(field, val) { state.roadmap[currentPillar][field] = val.trim(); saveState(); showToast("Pelan hidup disimpan."); }
    function addRoadmapItem() {
      const inp = document.getElementById("roadmapItemInput");
      if (!inp.value.trim()) return;
      state.roadmap[currentPillar].items.push({ text: inp.value.trim(), done: false });
      inp.value = "";
      saveState();
      renderRoadmap();
    }
    function toggleRoadmapItem(idx) { state.roadmap[currentPillar].items[idx].done = !state.roadmap[currentPillar].items[idx].done; saveState(); renderRoadmap(); }
    function deleteRoadmapItem(idx) { state.roadmap[currentPillar].items.splice(idx, 1); saveState(); renderRoadmap(); }

    /* MUHASABAH */
    function saveMuhasabah() {
      const win = document.getElementById("muhWin").value.trim();
      const imp = document.getElementById("muhImprove").value.trim();
      if (!win && !imp) return showToast("Tulis sekurang-kurangnya satu catatan ringkas.");
      state.muhasabahHistory.unshift({
        date: new Date().toLocaleDateString("ms-MY", { weekday: "short", day: "numeric", month: "short", year: "numeric" }),
        win: win || "-",
        improve: imp || "-"
      });
      document.getElementById("muhWin").value = "";
      document.getElementById("muhImprove").value = "";
      saveState();
      renderMuhasabahHistory();
      showToast("Alhamdulillah. Rehatlah dengan hati yang lapang malam ini.");
    }

    function renderMuhasabahHistory() {
      const box = document.getElementById("muhasabahHistoryList");
      document.getElementById("historyProTag").textContent = state.isPro ? "Penuh" : "⭐ Pro";
      if (state.muhasabahHistory.length === 0) {
        box.innerHTML = `<p class="muted">Belum ada catatan muhasabah. Mulakan malam ini — 30 saat sahaja.</p>`;
        return;
      }
      const listToShow = state.isPro ? state.muhasabahHistory : state.muhasabahHistory.slice(0, 2);
      let html = listToShow.map(item => `
        <div class="history-item">
          <div class="history-date">🌙 ${escapeHtml(item.date)}</div>
          <div><strong>Kejayaan kecil:</strong> ${escapeHtml(item.win)}</div>
          <div class="muted" style="margin-top:3px;"><strong>Baiki esok:</strong> ${escapeHtml(item.improve)}</div>
        </div>
      `).join("");

      if (!state.isPro && state.muhasabahHistory.length > 2) {
        html += `
          <div class="pro-lock-box mt-8">
            <h4>📓 Arkib Penuh Muhasabah (${state.muhasabahHistory.length} catatan)</h4>
            <p>Buka versi Pro untuk melihat semula semua perjalanan perubahan diri awak.</p>
            <button class="btn btn-gold btn-sm" onclick="openModal('proModal')">Buka Arkib Penuh</button>
          </div>
        `;
      }
      box.innerHTML = html;
    }

    /* PRO & BACKUP */
    function renderProModal() {
      document.getElementById("paymentLinkBtn").href = APP_CONFIG.paymentUrl;
      document.getElementById("proUnlockedNotice").style.display = state.isPro ? "block" : "none";
      document.getElementById("proPurchaseArea").style.display = state.isPro ? "none" : "block";

    function activateProCode() {
      const code = document.getElementById("proCodeInput").value.trim().toUpperCase();
      if (!code) return showToast("Sila masukkan kod akses Pro.");
      if (APP_CONFIG.validProCodes.includes(code)) {
        state.isPro = true;
        saveState();
        renderAll();
        renderProModal();
        playSoftClick(780, 0.2);
        showToast("🎉 Tahniah! Versi Pro berjaya diaktifkan.");
      } else { showToast("Kod tidak sah. Semak semula ejaan kod anda."); }
    }

    function exportDataJSON() {
      if (!state.isPro) { closeModal("settingsModal"); openModal("proModal"); return showToast("Backup JSON tersedia untuk ahli Pro."); }
      const blob = new Blob([JSON.stringify(state, null, 2)], { type: "application/json" });
      const url = URL.createObjectURL(blob);
      const a = document.createElement("a");
      a.href = url;
      a.download = `LangkahBaru_Backup_${getTodayKey()}.json`;
      a.click();
      URL.revokeObjectURL(url);
      showToast("Fail backup berjaya dimuat turun.");
    }

    function importDataJSON(e) {
      const file = e.target.files[0];
      if (!file) return;
      const reader = new FileReader();
      reader.onload = () => {
        try {
          state = { ...defaultState(), ...JSON.parse(reader.result) };
          saveState();
          renderAll();
          closeModal("settingsModal");
          showToast("Data backup berjaya dipulihkan!");
        } catch (err) { showToast("Fail backup tidak sah."); }
      };
      reader.readAsText(file);
    }

    function resetAllData() {
      if (!confirm("Pasti mahu padam semua data dan mula semula?")) return;
      localStorage.removeItem(APP_CONFIG.storageKey);
      state = defaultState();
      saveState();
      renderAll();
      closeModal("settingsModal");
      showToast("Semua data ditetapkan semula.");
    }

    function escapeHtml(str) {
      return String(str || "").replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;").replace(/"/g, "&quot;");
    }

      proBtn.classList.toggle("pro-active", state.isPro);

    }
